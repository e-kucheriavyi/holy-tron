TRON | larana.tech

Play

Rules

  • Player can move left, right, up or down.
  • Player moves every tick.
  • Player leaves trace behind.
  • Colliding with any trace kills player.
  • Leaving boundaries of a map kills player.
  • Last alive player wins.

Network

  1. Connect to WebSocket (endpoint is below).
  2. Send your nickname (if phase is PENDING).
  3. Wait for game to start.
  4. Control your player by sending single character: "l" | "r" | "u" | "d" (see Direction in types below).

WebSocket endpoint

wss://tron.larana.tech/ws

Types

const Direction = {
	LEFT:  "l",
	RIGHT: "r",
	UP:    "u",
	DOWN:  "d",
} as const

type Direction = (typeof Direction)[keyof typeof Direction]

const Phase = {
	PENDING: "p",
	READY:   "r",
	GAME:    "g",
	WINNER:  "w",
	END:     "e",
} as const

type Phase = (typeof Phase)[keyof typeof Phase]

type Player = {
	id:        number
	x:         number
	y:         number
	nickname:  string
	color:     string
	direction: Direction
	isAlive:   boolean
	connected: boolean
	trace:     number[]
	games:     number
	wins:      number
}

type ServerMessage = {
	/* timestamp */
	t: number
	/* tick number */
	i: number
	/* list of actions */
	a: Action[]
}

type WelcomeData = {
	// player id
	i: number
	// game id
	g: number
	// tick interval
	s: number
	// cells per player: arenaSize = a * players.length
	a: number
	p: PlayerConnectedData[]
}

type TurnData = {
	// player id
	i: number
	// new direction
	d: Direction
}

type PhaseData = {
	p: Phase
	// next phase interval
	n: number
	// winner id
	w: number
}

type PlayerConnectedData = {
	// id
	i: number
	// nickname
	n: string
	// color
	c: string
	// direction
	d: Direction
	x: number
	y: number
	// games played
	g: number
	// wins
	w: number
}

const ActionType = {
	WELCOME:          "w",
	PLAYER_CONNECTED: "p",
	PLAYER_DIED:      "d",
	PLAYER_TURNED:    "t",
	GAME_PHASE:       "h",
	DATA:             "r",
} as const

type ActionType = (typeof ActionType)[keyof typeof ActionType]

type Action = {
	t: ActionType
	// data
	d: unknown
}
